#include <stdio.h>

#include <stdlib.h>

struct Node {

     int data;

     struct Node* next;

};

struct Node* top = NULL;

struct Node* createNode(int val) {

     struct Node* newNode = (struct Node*)malloc(sizeof(struct Node));

     newNode->data = val;

     newNode->next = NULL;

     return newNode;

}

void push(int val) {

     struct Node* newNode = createNode(val);





    newNode->next = top;

    top = newNode;

    printf("Pushed: %d\n", val);

}

void pop() {

    if (top == NULL) {

        printf("Stack Underflow! Stack is empty.\n");

        return;

    }

    struct Node* temp = top;

    printf("Popped: %d\n", top->data);

    top = top->next;

    free(temp);

}

void peek() {

    if (top == NULL) {

        printf("Stack is Empty.\n");

        return;

    }

    printf("Top element: %d\n", top->data);

}

void display() {

    if (top == NULL) {

        printf("Stack is Empty.\n");

        return;

    }




    struct Node* temp = top;

    printf("Stack (Top -> Bottom): ");

    while (temp != NULL) {

      printf("%d ", temp->data);

      temp = temp->next;
    }
    printf("\n");
}
int main() {
    push(10);
    push(20);
    push(30);
    push(40);
    display();
    peek();
    pop();
    pop();
    display();
    peek();
    return 0;
}
